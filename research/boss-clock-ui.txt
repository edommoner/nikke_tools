function renderPhases(){
 $('enemy-phase-list').innerHTML=state.enemyPatterns.map((p,i)=>{const v=editablePhase(p);if(!v)return `<div class="phase-row"><p>取り込み済みの行動：${e(p.note||p.kind)}（詳細定義を保持）</p><button class="button text" data-remove-phase="${i}">削除</button></div>`;return `<div class="phase-row" data-phase-row="${i}"><label>行動<select data-phase-field="kind" aria-label="行動${i+1}の種類"><option value="core" ${v.kind==='core'?'selected':''}>コア出現</option><option value="move" ${v.kind==='move'?'selected':''}>距離変更</option></select></label><label>${v.kind==='core'?'コア径（px）':'距離（m）'}<input data-phase-field="value" aria-label="行動${i+1}の値" type="number" min="${v.kind==='core'?1:0}" max="1000" step="${v.kind==='core'?1:.1}" value="${v.value}"></label><label>開始まで待つ（秒）<input data-phase-field="wait" aria-label="行動${i+1}の待ち時間" type="number" min="0" max="180" step="0.1" value="${v.wait}"></label><label>継続時間（秒）<input data-phase-field="duration" aria-label="行動${i+1}の継続時間" type="number" min="0.01" max="180" step="0.1" value="${v.duration}"></label><label class="check"><input data-phase-field="repeat" type="checkbox" aria-label="行動${i+1}を繰り返す" ${v.repeat?'checked':''}>繰り返す</label><button class="button text" data-remove-phase="${i}">削除</button></div>`;}).join('');
}
function phaseChanged(){renderPhases();$('enemy-preset').value='';rememberEnemy(enemyScope(state.singleMode,state.raidCount));renderBoard();}
$('add-enemy-phase').onclick=guarded(()=>{requireEditable();if(state.enemyPatterns.length>=32)throw Error('この画面で作成できる行動は32件までです。');state.enemyPatterns.push(makePhase({id:'phase-'+crypto.randomUUID(),kind:'core',wait:3,duration:5,value:6,repeat:false}));phaseChanged();});
$('enemy-phase-list').onchange=guarded(event=>{requireEditable();const input=event.target,key=input.dataset.phaseField;if(!key)return;const i=Number(input.closest('[data-phase-row]').dataset.phaseRow),value=editablePhase(state.enemyPatterns[i]);if(!value)throw Error('この行動は取り込み元の定義で編集してください。');value[key]=key==='repeat'?input.checked:key==='kind'?input.value:input.valueAsNumber;if(key==='kind')value.value=value.kind==='core'?6:num('distance');state.enemyPatterns[i]=makePhase(value);phaseChanged();});
$('enemy-phase-list').onclick=guarded(event=>{const button=event.target.closest('[data-remove-phase]');if(!button)return;requireEditable();state.enemyPatterns.splice(Number(button.dataset.removePhase),1);phaseChanged();});
function bossDetails(result){const rows=result.boss_timeline||[];if(!result.boss_patterns?.length)return '';const names={core:'コア出現',move:'距離変更',idle:'待機',groggy:'ダウン区間',buff:'敵の能力変化',parts:'パーツ',interrupt:'阻止部位',attack:'攻撃',debuff:'弱体化',summon:'増援',shield:'バリア',vanish:'姿を隠す'},outcomes={expired:'時間終了',followed:'別の行動に続く',cleared:'対象の消失',end:'計算終了'};return `<details class="boss-details"><summary>ボス行動の開始・終了を見る（${rows.length}件）</summary><p class="subtle">設定した行動の実行履歴です。${rows.length>100?'先頭100件を表示しています。全件は保存・書き出しに含みます。':''}</p><div class="table-scroll"><table><thead><tr><th>時刻</th><th>行動</th><th>状態</th><th>終了理由</th></tr></thead><tbody>${rows.slice(0,100).map(r=>`<tr><td>${Number(r.time).toFixed(2)}秒</td><td>${e(result.boss_patterns.find(p=>p.id===r.pattern)?.note||names[r.kind]||r.kind)}</td><td>${r.event==='start'?'開始':'終了'}</td><td>${e(outcomes[r.outcome]||'—')}</td></tr>`).join('')||'<tr><td colspan="4">この計算時間内に開始した行動はありません。</td></tr>'}</tbody></table></div></details>`;}
